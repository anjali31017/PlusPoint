  <script>


    // Example validation + AJAX
    $(document).ready(function () {

      // --- 1. Auto-redirect if already logged in ---
      function checkSession() {
        $.ajax({
          url: 'http://127.0.0.1:5000/api/admin/check-session',
          method: 'GET',
          xhrFields: { withCredentials: true },
          timeout: 3000,
          success: function (res) {
            if (res.status === 1 && res.admin_id) {
              window.location.replace('kyc.html');
            }
          }
        });
      }


      setTimeout(checkSession, 100);


      $('#loginForm').submit(function (e) {
        e.preventDefault();
        $('#emailError, #passwordError').hide();

        let email = $('#email').val().trim();
        let password = $('#password').val().trim();
        let valid = true;

        if (email === '') {
          $('#emailError').text('Email is required').show();
          valid = false;
        } else if (!/^\S+@\S+\.\S+$/.test(email)) {
          $('#emailError').text('Enter a valid email').show();
          valid = false;
        }

        if (password === '') {
          $('#passwordError').text('Password is required').show();
          valid = false;
        } else if (password.length < 6) {
          $('#passwordError').text('Password must be at least 6 characters long').show();
          valid = false;
        }

        if (!valid) return;

        // Replace with your API endpoint


        // --- AJAX POST to login ---
        $.ajax({
          url: 'http://127.0.0.1:5000/api/admin/login',
          method: 'POST',
          contentType: 'application/json',
          data: JSON.stringify({ email: email, password: password }),
          success: function (res) {
            if (res.status === 1) {
              window.location.href = 'kyc.html';

            } else {
              Swal.fire({
                icon: 'error',
                title: 'Oops...',
                text: res.message
              });
            }
          },
          error: function (xhr) {
            let msg = "Something went wrong";
            if (xhr.responseJSON && xhr.responseJSON.detail) {
              msg = xhr.responseJSON.detail;
            }
            Swal.fire({
              icon: 'error',
              title: 'Login Failed',
              text: msg
            });
          }
        });
      });

    });
  </script>
